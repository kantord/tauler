// Scenario: a solid-colour wallpaper and one fixed-width left panel on top of
// it, with one static d3 chart in the lower part of the panel.
//
// Backend-neutral on purpose: it has no <I3Layout>, no module and no client, so
// the same file runs on the i3 and the Sway desktops and the two screenshots
// can be compared. The expected colours live in tauler-e2e/tests/scenarios.rs,
// written out by hand.
//
// The chart is there so that comparison covers more than flat fills: axes with
// text labels, gridlines, bars and a line path, all through d3's scale, axis
// and shape APIs. Its data is fixed and nothing animates, so both desktops must
// draw the same pixels. d3 and linkedom are fetched on first use (ADR 0045), so
// the chart appears some seconds after the panel does.
//
// The chart is a 240x400 figure, 16px from the panel's left edge and 64px from
// the bottom (y 616..1016). The panel's vertical middle (y 540), its corners
// and its right-most column stay flat panel colour, because the scenarios probe
// them. The first bar is the tallest, so it runs the whole height of the plot
// and its colour (#f9e2af) is a safe probe: the test reads it at the figure's
// pixel (83, 170), i.e. screen (99, 786), which no gridline or label touches.

const WIDTH = 240;
const HEIGHT = 400;
const MARGIN = { top: 8, right: 8, bottom: 24, left: 40 };

const DATA = [
  { name: "mon", value: 8 },
  { name: "tue", value: 5 },
  { name: "wed", value: 6.5 },
  { name: "thu", value: 3 },
];

function Chart() {
  return (
    <D3Figure
      key="chart"
      width={WIDTH}
      height={HEIGHT}
      draw={(svg, d3) => {
        const x = d3.scaleBand()
          .domain(DATA.map((d) => d.name))
          .range([MARGIN.left, WIDTH - MARGIN.right])
          .padding(0.2);
        const y = d3.scaleLinear()
          .domain([0, 8])
          .range([HEIGHT - MARGIN.bottom, MARGIN.top]);

        // Gridlines first, so the bars cover them.
        svg.append("g")
          .selectAll("line")
          .data(y.ticks(4))
          .join("line")
          .attr("x1", MARGIN.left)
          .attr("x2", WIDTH - MARGIN.right)
          .attr("y1", (t) => y(t))
          .attr("y2", (t) => y(t))
          .attr("stroke", "#585b70")
          .attr("stroke-width", 1);

        svg.append("g")
          .selectAll("rect")
          .data(DATA)
          .join("rect")
          .attr("x", (d) => x(d.name))
          .attr("width", x.bandwidth())
          .attr("y", (d) => y(d.value))
          .attr("height", (d) => y(0) - y(d.value))
          .attr("fill", "#f9e2af");

        svg.append("path")
          .attr("fill", "none")
          .attr("stroke", "#89b4fa")
          .attr("stroke-width", 2)
          .attr("d", d3.line()
            .x((d) => x(d.name) + x.bandwidth() / 2)
            .y((d) => y(d.value))(DATA));

        svg.append("g")
          .attr("transform", `translate(0,${HEIGHT - MARGIN.bottom})`)
          .call(d3.axisBottom(x))
          .call((g) => g.selectAll("path, line").attr("stroke", "#cdd6f4"))
          .call((g) => g.selectAll("text").attr("fill", "#cdd6f4").attr("font-size", 12));

        svg.append("g")
          .attr("transform", `translate(${MARGIN.left},0)`)
          .call(d3.axisLeft(y).ticks(4))
          .call((g) => g.selectAll("path, line").attr("stroke", "#cdd6f4"))
          .call((g) => g.selectAll("text").attr("fill", "#cdd6f4").attr("font-size", 12));
      }}
    />
  );
}

export default function render() {
  return (
    <root>
      <wallpaper id="desktop">
        <div style={{ width: "100%", height: "100%", backgroundColor: "#cc3366" }} />
      </wallpaper>
      <panel id="rail" anchor="left" width={272} height={1080}>
        <div style={{ width: "100%", height: "100%", backgroundColor: "#1e1e2e", display: "flex", flexDirection: "column", justifyContent: "flex-end", alignItems: "flex-start", paddingLeft: 16, paddingBottom: 64 }}>
          <Chart />
        </div>
      </panel>
    </root>
  );
}
