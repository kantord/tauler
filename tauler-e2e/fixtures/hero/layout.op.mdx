const BAR =
  "flex h-full items-center gap-3 px-3 bg-background text-foreground";

function Bar({ time }) {
  return (
    <Module bin="~/.cargo/bin/tauler-i3">
      {(data) => (
        <div class={BAR}>
          {(data?.workspaces ?? []).map((ws) => (
            <span class={ws.focused ? "" : "opacity-60"}>
              {ws.name}
            </span>
          ))}
          <span>{time}</span>
        </div>
      )}
    </Module>
  );
}

export default function render() {
  const time = useStringStream("/bin/sh",
    "while :; do date +%H:%M; sleep 1; done");
  return (
    <root>
      <I3Layout module="~/.cargo/bin/tauler-i3">
        <Panel id="bar" anchor="top" size={28}>
          <Bar time={time} />
        </Panel>
      </I3Layout>
      <dom><Bar time={time} /></dom>
    </root>
  );
}
